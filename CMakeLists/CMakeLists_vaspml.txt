# CMakeLists for VASPml - machine learning library for VASP (experimental)
# Builds libvaspml (linked into vasp_std/gam/ncl) and standalone tool
# executables. Activated via -DVASP_VASPML=ON.

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# MPI CXX is found by the root CMakeLists when VASP_VASPML=ON.
if(NOT TARGET MPI::MPI_CXX)
  message(
    FATAL_ERROR
      "VASPml requires MPI CXX but MPI::MPI_CXX was not found. "
      "Ensure VASP_VASPML=ON is set so the root CMakeLists finds MPI CXX.")
endif()

# -----------------------------------------------------------------------
# libvaspml static library
# -----------------------------------------------------------------------
file(GLOB VASPML_LIB_SOURCES CONFIGURE_DEPENDS
     ${CMAKE_CURRENT_SOURCE_DIR}/src/libvaspml/*.cpp)

add_library(vaspml STATIC ${VASPML_LIB_SOURCES})
target_compile_features(vaspml PUBLIC cxx_std_17)
target_include_directories(vaspml
                           PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src/libvaspml)
target_link_libraries(vaspml PUBLIC MPI::MPI_CXX BLAS::BLAS)

# Propagate target CPU architecture to C++ VASPml library and standalone tools
# when explicitly set (matching fftlib/parser, without forcing -march=native
# when unset so external wrappers such as Spack stay in control).
if(VASP_TARGET_CPU)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU"
     OR CMAKE_CXX_COMPILER_ID MATCHES "Clang"
     OR CMAKE_CXX_COMPILER_ID MATCHES "Intel")
    target_compile_options(
      vaspml PUBLIC $<$<COMPILE_LANGUAGE:CXX>:-march=${VASP_TARGET_CPU}>)
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "NVHPC")
    target_compile_options(
      vaspml PUBLIC $<$<COMPILE_LANGUAGE:CXX>:-tp=${VASP_TARGET_CPU}>)
  endif()
endif()

# std::filesystem lives in a separate library (stdc++fs) on older toolchains
# (GCC < 9). NVHPC and other compilers may use such a toolchain as their host
# C++ standard library.
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_STANDARD 17)
check_cxx_source_compiles(
  "
    #include <filesystem>
    int main() { return std::filesystem::exists(\"/tmp\") ? 0 : 1; }
"
  VASPML_STDFS_LINKS)
if(NOT VASPML_STDFS_LINKS)
  if(NOT VASPML_MESSAGE_SHOWN)
    message(STATUS "VASPml: linking stdc++fs for std::filesystem support")
  endif()
  target_link_libraries(vaspml PUBLIC stdc++fs)
endif()

# Auto-detect whether MKL CBLAS headers are needed. Intel compilers define
# VASPML_USE_MKL themselves via __INTEL_COMPILER/__INTEL_LLVM_COMPILER in
# Linalg.hpp, so no flag is needed there. For non-Intel compilers building
# against MKL (e.g. GNU + MKL), we detect this and set VASPML_USE_MKL
# explicitly.
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Intel|IntelLLVM")
  set(_vaspml_mkl FALSE)
  if(BLA_VENDOR MATCHES "Intel")
    set(_vaspml_mkl TRUE)
  elseif(VASP_NVHPC_MKL)
    set(_vaspml_mkl TRUE)
  elseif(BLAS_LIBRARIES MATCHES "mkl")
    set(_vaspml_mkl TRUE)
  elseif(DEFINED ENV{MKLROOT} OR DEFINED ENV{ONEAPI_ROOT})
    set(_vaspml_mkl TRUE)
  endif()
  if(_vaspml_mkl)
    if(NOT VASPML_MESSAGE_SHOWN)
      message(
        STATUS
          "VASPml: non-Intel compiler with MKL detected, enabling VASPML_USE_MKL"
      )
    endif()
    target_compile_definitions(vaspml PUBLIC VASPML_USE_MKL)
  endif()
endif()

# Locate the CBLAS header directory. MKL: mkl_cblas.h lives under
# $MKLROOT/include; OpenBLAS: cblas.h under $OPENBLAS_ROOT/include.
find_path(
  CBLAS_INCLUDE_DIR
  NAMES mkl_cblas.h cblas.h
  HINTS $ENV{MKLROOT} $ENV{ONEAPI_ROOT}/mkl/latest ${BLAS_ROOT} $ENV{BLAS_ROOT}
        $ENV{OPENBLAS_ROOT}
  PATH_SUFFIXES include)
if(CBLAS_INCLUDE_DIR)
  target_include_directories(vaspml PRIVATE ${CBLAS_INCLUDE_DIR})
else()
  if(NOT VASPML_MESSAGE_SHOWN)
    message(
      WARNING
        "VASPml: CBLAS include directory not found. Set MKLROOT, OPENBLAS_ROOT, or BLAS_ROOT."
    )
  endif()
endif()

set(VASPML_MESSAGE_SHOWN
    TRUE
    CACHE INTERNAL "VASPml messages shown flag")

# -----------------------------------------------------------------------
# GRACE force field support (optional, requires TensorFlow + cppflow)
# -----------------------------------------------------------------------
set(_tf_rpath_dirs "")
if(VASP_VASPML_GRACE)
  target_compile_definitions(vaspml PUBLIC VASPML_ENABLE_GRACE)
  find_package(TensorFlow MODULE REQUIRED)
  target_link_libraries(vaspml PUBLIC TensorFlow::tensorflow_cc)
  if(TARGET TensorFlow::tensorflow_framework)
    target_link_libraries(vaspml PUBLIC TensorFlow::tensorflow_framework)
  endif()
  # cppflow is a header-only library; locate its include dir
  find_path(
    CPPFLOW_INCLUDE_DIR
    NAMES cppflow/cppflow.h cppflow/model.h
    HINTS ${CPPFLOW_ROOT} $ENV{CPPFLOW_ROOT} ${TENSORFLOW_ROOT}
          $ENV{TENSORFLOW_ROOT}
    PATH_SUFFIXES include cppflow/include)
  if(CPPFLOW_INCLUDE_DIR)
    target_include_directories(vaspml PUBLIC ${CPPFLOW_INCLUDE_DIR})
    message(
      STATUS "VASPml GRACE: using cppflow headers from ${CPPFLOW_INCLUDE_DIR}")
  else()
    message(
      FATAL_ERROR
        "VASPml GRACE: cppflow headers (cppflow/cppflow.h) not found. Set CPPFLOW_ROOT."
    )
  endif()
  # TensorFlow version (informational; cppflow generally needs >= 2.x)
  set(_tf_version "")
  if(EXISTS "${TENSORFLOW_INCLUDE_DIRS}/tensorflow/core/public/version_macros.h")
    file(STRINGS "${TENSORFLOW_INCLUDE_DIRS}/tensorflow/core/public/version_macros.h"
         _tf_version_def REGEX "^#define +TF_VERSION_STRING")
    if(_tf_version_def MATCHES "\"([0-9]+\\.[0-9]+\\.[0-9]+[^\"]*)\"")
      set(_tf_version "${CMAKE_MATCH_1}")
    endif()
  endif()
  if(_tf_version)
    if(_tf_version VERSION_LESS "2.0.0")
      message(WARNING "VASPml GRACE: TensorFlow ${_tf_version} is very old; "
                      "cppflow expects at least the 2.x C API.")
    else()
      message(STATUS "VASPml GRACE: TensorFlow ${_tf_version}")
    endif()
  else()
    message(STATUS "VASPml GRACE: TensorFlow version unknown (no version_macros.h)")
  endif()
  # Propagate TensorFlow runtime library directory to executables' RPATH so no
  # LD_LIBRARY_PATH is needed at runtime (e.g. py-tensorflow installs into
  # site-packages/tensorflow/ outside standard lib/). vaspml is STATIC so RPATH
  # must be attached to final executable targets (here + parent src/ scope).
  foreach(_tf_tgt TensorFlow::tensorflow_cc TensorFlow::tensorflow_framework)
    if(TARGET ${_tf_tgt})
      get_target_property(_tf_loc ${_tf_tgt} IMPORTED_LOCATION)
      if(_tf_loc)
        get_filename_component(_tf_dir "${_tf_loc}" DIRECTORY)
        list(APPEND _tf_rpath_dirs "${_tf_dir}")
      endif()
    endif()
  endforeach()
  list(REMOVE_DUPLICATES _tf_rpath_dirs)
  if(_tf_rpath_dirs)
    set(VASPML_RPATH_DIRS
        "${_tf_rpath_dirs}"
        PARENT_SCOPE)
  endif()
  message(STATUS "VASPml GRACE: enabled (TensorFlow + cppflow)")
endif()

install(TARGETS vaspml ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})

# -----------------------------------------------------------------------
# VASPml standalone application executables
# -----------------------------------------------------------------------
file(GLOB VASPML_APP_SOURCES CONFIGURE_DEPENDS
     ${CMAKE_CURRENT_SOURCE_DIR}/src/applications/*.cpp)

foreach(app_source ${VASPML_APP_SOURCES})
  get_filename_component(app_name ${app_source} NAME_WE)
  add_executable(${app_name} ${app_source})
  target_link_libraries(${app_name} PRIVATE vaspml MPI::MPI_CXX)
  if(_tf_rpath_dirs)
    set_property(TARGET ${app_name} APPEND PROPERTY BUILD_RPATH ${_tf_rpath_dirs})
    set_property(TARGET ${app_name} APPEND PROPERTY INSTALL_RPATH ${_tf_rpath_dirs})
  endif()
  install(TARGETS ${app_name} DESTINATION ${CMAKE_INSTALL_BINDIR})
endforeach()
